# Builds the deploy binaries on ubuntu-22.04 (glibc 2.35, same floor as the jammy rule) and attaches them to a release.
# Trigger: push a tag v* or run by hand. No secrets used.
name: release-binaries
on:
  workflow_dispatch:
  push:
    tags: ['v*']
permissions:
  contents: write
jobs:
  build:
    runs-on: ubuntu-22.04
    steps:
      - uses: actions/checkout@v4
      - uses: dtolnay/rust-toolchain@stable
      - run: sudo apt-get update && sudo apt-get install -y libssl-dev pkg-config build-essential
      - run: cargo build --release -p mail4agent-server-bin -p m4a-edge
      - name: glibc ceiling
        run: |
          for b in mail4agent-server-bin m4a-edge; do objdump -T target/release/$b | grep -o 'GLIBC_[0-9.]*' | sort -Vu | tail -1; done
      - name: package
        run: |
          mkdir dist && cp target/release/mail4agent-server-bin target/release/m4a-edge dist/
          cp -r deploy dist/deploy
          tar -C dist -czf m4a-binaries-${GITHUB_SHA::8}.tar.gz .
          sha256sum m4a-binaries-*.tar.gz > SHA256SUMS
      - uses: actions/upload-artifact@v4
        with: { name: m4a-binaries, path: "m4a-binaries-*.tar.gz\nSHA256SUMS" }
      - if: startsWith(github.ref, 'refs/tags/')
        uses: softprops/action-gh-release@v2
        with: { files: "m4a-binaries-*.tar.gz\nSHA256SUMS" }
